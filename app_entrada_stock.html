function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.openById(data.sheetId);
  const ws = ss.getSheetByName(data.sheetName);
  ws.appendRow(data.row);
  return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  const ss = SpreadsheetApp.openById(e.parameter.sheetId);
  const ws = ss.getSheetByName(e.parameter.sheetName || 'Taules_Stock');
  const rows = ws.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify(rows)).setMimeType(ContentService.MimeType.JSON);
}
